function renderStreamingLinks(anime) { const list = document.getElementById("episodeList"); list.innerHTML = ""; const episodes = anime.streamingEpisodes || []; if (episodes.length > 0) { episodes.forEach((episode, index) => { if (!episode.url) return; const row = document.createElement("div"); row.className = "episode"; const details = document.createElement("div"); const title = document.createElement("strong"); title.textContent = episode.title || `Episode ${index + 1}`; const site = document.createElement("small"); site.textContent = episode.site ? `Available on ${episode.site}` : "External streaming link"; details.append(title, site); const button = document.createElement("button"); button.textContent = "▶ Watch"; button.onclick = () => { const url = safeURL(episode.url); if (!url) { alert("This streaming link is not supported."); return; } window.open(url, "_blank", "noopener,noreferrer"); }; row.append(details, button); list.append(row); }); } if (!list.children.length) { const message = document.createElement("div"); message.className = "status"; message.textContent = "No episode links are listed for this anime yet."; list.append(message); if ( anime.trailer && anime.trailer.site === "youtube" && anime.trailer.id ) { const trailerButton = document.createElement("button"); trailerButton.textContent = "▶ Watch Trailer"; trailerButton.style.marginTop = "12px"; trailerButton.onclick = () => { window.open( "https://www.youtube.com/watch?v=" + encodeURIComponent(anime.trailer.id), "_blank", "noopener,noreferrer" ); }; list.append(trailerButton); } } }
